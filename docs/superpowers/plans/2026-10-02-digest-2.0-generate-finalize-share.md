# Digest 2.0 Generate → Finalize → Share Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let Matt generate a round's 2.0 digest from the app, finalize it, and share a frozen static copy at `https://digest.mattmariani.com/d2/<slug>` that never exposes the admin host.

**Architecture:** The existing read-only `/digest2/[roundId]` page gains a share mode (bare layout, no admin chrome) and an admin bar (Generate → existing 1.0 draft endpoint; Finalize → new publish endpoint). A forked publish renderer (Puppeteer, modelled on 1.0's `renderDigestHtml`) captures `/digest2/<id>?share=1` plus its `/_app/` asset closure, rewrites it for static serving, runs a hard leak scan, and atomically writes `digests/d2/<slug>/`. Caddy's digest host gains a static `/d2/*` block.

**Tech Stack:** SvelteKit (Svelte 5 runes), better-sqlite3, puppeteer-core + baked Chromium, Vitest, Caddy, docker compose.

**Spec:** `docs/superpowers/specs/2026-10-02-digest-2.0-generate-finalize-share-design.md`

## Global Constraints

- Players must never see the admin host: no occurrence of `new URL(appBaseUrl()).host` in any published file.
- 2.0 writes ONLY `digest2_shares` and `digests/d2/`. Never write `digest_*` (1.0) tables, `digests/d/`, or `digests/_media/`. Do **not** edit `ui/src/lib/digest/export.ts`.
- No hardcoded hostnames — use `appBaseUrl()` / `digestBaseUrl()` from `$lib/config/urls.js`. `src/lib/config/hostLiterals.test.ts` must not gain new failures (it has 3 pre-existing ones in untouched files).
- `Caddyfile.digest`: add only a static `/d2/*` block; never touch or widen the `/api/p/*` proxy.
- Slugs: `randomBytes(12).toString('base64url')` (16 chars, same as 1.0), one per round, stable forever (re-publish overwrites in place).
- Repo rules: gitnexus `impact` before editing an existing symbol (`node .gitnexus/run.cjs impact "<sym>" --direction upstream --repo .`, warn on HIGH/CRITICAL); `detect-changes --scope all` before each commit; `cd ui && npm run check` must stay at 0 errors.

import { randomBytes } from 'node:crypto';
import type Database from 'better-sqlite3';
import { digestBaseUrl } from '$lib/config/urls.js';

/** Public link for a published 2.0 digest — always the digest host, never the app host. */
export function digest2ShareUrl(slug: string): string {
	return `${digestBaseUrl()}/d2/${slug}`;
}

/** Stable, unguessable slug per round (one forever; re-publish overwrites in place). */
export function getOrCreateDigest2Slug(db: Database.Database, roundId: number): string {
	const row = db.prepare('SELECT slug FROM digest2_shares WHERE round_id = ?').get(roundId) as
		| { slug: string }
		| undefined;
	if (row?.slug) return row.slug;
	const slug = randomBytes(12).toString('base64url');
	db.prepare(
		`INSERT INTO digest2_shares (round_id, slug) VALUES (?, ?) ON CONFLICT(round_id) DO NOTHING`
	).run(roundId, slug);
	return (db.prepare('SELECT slug FROM digest2_shares WHERE round_id = ?').get(roundId) as { slug: string }).slug;
}

export function getDigest2Share(
	db: Database.Database,
	roundId: number
): { slug: string; updatedAt: string; url: string } | null {
	const row = db
		.prepare('SELECT slug, updated_at AS updatedAt FROM digest2_shares WHERE round_id = ?')
		.get(roundId) as { slug: string; updatedAt: string } | undefined;
	return row ? { ...row, url: digest2ShareUrl(row.slug) } : null;
}

export function touchDigest2Share(db: Database.Database, roundId: number): void {
	db.prepare(
		`UPDATE digest2_shares SET updated_at = strftime('%Y-%m-%dT%H:%M:%SZ','now') WHERE round_id = ?`
	).run(roundId);
}
```

// Pure HTML rules for the static 2.0 share artifact (forked from 1.0's
// transformShareHtml in lib/digest/export.ts — that file is not edited).

const FETCH_GUARD =
	`<script>(function(){var f=window.fetch;window.fetch=function(u){try{` +
	`var s=typeof u==="string"?u:(u&&u.url)||"";` +
	`if(s.indexOf("/api/ml-auth")!==-1)return Promise.resolve(new Response(null,{status:204}));` +
	`}catch(e){}return f.apply(this,arguments);};})();</script>`;

/** Rewrite a captured SSR document so it works from digests/d2/<slug>/ on a dumb static host. */
export function transformDigest2Html(html: string, opts: { hasLocalFonts: boolean }): string {
	let h = html;
	// ./ prefix is mandatory — a bare "_app/…" specifier is illegal for dynamic import().
	h = h.replaceAll('../_app/', './_app/').replaceAll('"/_app/', '"./_app/');
	h = h.replace(/[ \t]*<link\b[^>]*\brel="(?:icon|apple-touch-icon|manifest)"[^>]*>\n?/g, '');
	h = h.replace(/[ \t]*<link\b[^>]*fonts\.(?:googleapis|gstatic)\.com[^>]*>\n?/g, '');
	if (opts.hasLocalFonts) {
		h = h.replace('</head>', '\t\t<link rel="stylesheet" href="./_app/fonts.css" />\n\t</head>');
	}
	h = h.replace('<head>', '<head>\n\t\t' + FETCH_GUARD);
	return h;
}

const ROOT_RELATIVE_ATTR = /\b(href|src|action|poster)="\/(?!\/)[^"]*"/g;

function snippet(text: string, at: number): string {
	return text.slice(Math.max(0, at - 60), at + 80).replace(/\s+/g, ' ');
}

/**
 * Hard gate before anything is written: the published copy must not mention the
 * admin/internal hosts anywhere, and its HTML must not link root-relative (those
 * resolve to digest.mattmariani.com/<app route> → 404, or worse, an app path).
 */
export function findLeaks(input: {
	html: string;
	textAssets: Map<string, string>;
	forbiddenHosts: string[];
}): string[] {
	const leaks: string[] = [];
	const sources: [string, string][] = [['index.html', input.html], ...input.textAssets];
	for (const [name, text] of sources) {
		for (const host of input.forbiddenHosts) {
			const at = text.indexOf(host);
			if (at !== -1) leaks.push(`${name}: forbidden host "${host}" … ${snippet(text, at)}`);
		}
	}
	for (const m of input.html.matchAll(ROOT_RELATIVE_ATTR)) {
		leaks.push(`index.html: root-relative ${m[0]} … ${snippet(input.html, m.index ?? 0)}`);
	}
	return leaks;
}

export interface Digest2Chrome {
	share: boolean;
	meta: { leagueName: string; roundName: string };
	/** null in share mode — nothing admin-only may reach the public copy. */
	admin: { draftGeneratedAt: string | null; published: { url: string; updatedAt: string } | null } | null;

export function buildDigest2Chrome(db: Database.Database, roundId: number, share: boolean): Digest2Chrome {
	const row = db
		.prepare(
			`SELECT l.name AS leagueName, r.name AS roundName
			   FROM rounds r JOIN seasons s ON s.id = r.season_id JOIN leagues l ON l.id = s.league_id
			  WHERE r.id = ?`
		)
		.get(roundId) as { leagueName: string; roundName: string } | undefined;
	const meta = row ?? { leagueName: '', roundName: '' };
	if (share) return { share, meta, admin: null };
	const draft = getActiveDraftForRound(db, roundId);
	const pub = getDigest2Share(db, roundId);
	return {
		share,
		meta,
		admin: {
			draftGeneratedAt: draft?.generated_at ?? null,
			published: pub ? { url: pub.url, updatedAt: pub.updatedAt } : null
		}
	};
}

/** Swap a fully-written tmp dir into place; the old copy survives any failure before the swap. */
export async function swapDirAtomically(tmpDir: string, finalDir: string): Promise<void> {
	await stat(join(tmpDir, 'index.html')); // throws if the new copy is incomplete
	const backup = `${finalDir}.old`;
	await rm(backup, { recursive: true, force: true });
	let hadOld = true;
	try { await rename(finalDir, backup); } catch { hadOld = false; }
	try {
		await rename(tmpDir, finalDir);
	} catch (e) {
		if (hadOld) await rename(backup, finalDir);
		throw e;
	}
	await rm(backup, { recursive: true, force: true });
}

// Forked from 1.0 export.ts localizeFonts (not imported: that module is 1.0-owned).
async function localizeFonts(html: string, assets: Map<string, Buffer>): Promise<string | null> {
	const m = html.match(/https:\/\/fonts\.googleapis\.com\/css2\?[^"'\s]+/);
	if (!m) return null;
	try {
		let css = await (await fetch(m[0].replace(/&amp;/g, '&'), { headers: { 'User-Agent': CHROME_UA } })).text();
		const urls = [...new Set([...css.matchAll(/https:\/\/fonts\.gstatic\.com\/[^)'"]+\.woff2/g)].map((x) => x[0]))];

		page.on('response', async (resp) => {
			try {
				if (resp.request().method() !== 'GET' || !resp.ok()) return;
				const u = new URL(resp.url());
				if (u.origin === origin && u.pathname.startsWith('/_app/')) assets.set(u.pathname, await resp.buffer());
			} catch { /* a single asset capture failure is non-fatal; the leak scan + integration check catch gaps */ }
		});
		const url = `${APP_INTERNAL_URL}/digest2/${roundId}?share=1`;
		const resp = await page.goto(url, { waitUntil: 'networkidle0', timeout: 45_000 });
		if (!resp || !resp.ok()) throw new Digest2PublishError(500, `render load failed: ${resp?.status() ?? 'no-response'}`);
		rawHtml = await resp.text();
	} catch (e) {
		if (e instanceof Digest2PublishError) throw e;
		throw new Digest2PublishError(500, `render failed: ${e instanceof Error ? e.message : String(e)}`);
	} finally {
		await browser.close().catch(() => {});
	}

	const fontCss = await localizeFonts(rawHtml, assets);
	const indexHtml = transformDigest2Html(rawHtml, { hasLocalFonts: !!fontCss });
	const textAssets = new Map<string, string>();
	for (const [p, buf] of assets) if (TEXT_EXT.has(extname(p))) textAssets.set(p, buf.toString('utf8'));
	const leaks = findLeaks({
		html: indexHtml,
		textAssets,
		forbiddenHosts: [new URL(appBaseUrl()).host, new URL(APP_INTERNAL_URL).host]
	});
	if (leaks.length) throw new Digest2PublishError(422, `Publish blocked — admin links found:\n${leaks.slice(0, 5).join('\n')}`);

	await rm(tmpDir, { recursive: true, force: true });
	await mkdir(tmpDir, { recursive: true });
	let bytes = 0, files = 0;
	const write = async (rel: string, data: Buffer | string) => {
		const dest = join(tmpDir, rel);
		await mkdir(dirname(dest), { recursive: true });
		await writeFile(dest, data);
		bytes += typeof data === 'string' ? Buffer.byteLength(data) : data.length;
		files += 1;
	};
	for (const [p, buf] of assets) await write(p.replace(/^\//, ''), buf);
	if (fontCss) await write('_app/fonts.css', fontCss);
	await write('index.html', indexHtml);
	await swapDirAtomically(tmpDir, finalDir);
	touchDigest2Share(db, roundId);
	return { slug, url: digest2ShareUrl(slug), files, bytes };
}
```

import type { RequestHandler } from './$types.js';
import { json, error } from '@sveltejs/kit';
import { getDb } from '$lib/db/client.js';
import { getActiveDraftForRound } from '$lib/digest/llm.js';

// POST /api/digest2/:roundId/publish — render the 2.0 digest to digests/d2/<slug>/
// and return its digest-host link. Writes only digest2_shares + digests/d2/.
export const POST: RequestHandler = async ({ params }) => {
	const roundId = Number(params.roundId);
	if (!Number.isInteger(roundId) || roundId <= 0) throw error(400, 'bad round id');
	const db = getDb();
	if (!db.prepare('SELECT 1 FROM rounds WHERE id = ?').get(roundId)) throw error(404, 'round not found');
	if (!getActiveDraftForRound(db, roundId)) throw error(409, 'No story draft yet — press Generate first.');
	try {
		return json(await publishDigest2(roundId));
	} catch (e) {
		if (e instanceof Digest2PublishError) throw error(e.status, e.message);
		throw error(500, `publish failed: ${e instanceof Error ? e.message : String(e)}`);
	}
};

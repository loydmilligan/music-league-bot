**Goal:** Restructure the read-only `/digest2/[roundId]` page into two top-level tabs — **Digest** and **Stats** — where the full ML Stats explorer lives in the Stats tab, and a single **Leaderboard Race** panel is teased in the Digest tab with a gateway link into the Stats tab (and is NOT duplicated in Stats).

Extract the self-contained "Leaderboard Race" section out of `LeaderboardTab.svelte` into a reusable `LeaderboardRaceChart.svelte` (used by both the tab and the Digest teaser). Add `omitTabs` + `omitLeaderboardRace` opt-in props to `StatsExplorer`/`LeaderboardTab` for the digest2 embedding. Add a top-level tab shell to the digest2 route.

**Tech Stack:** SvelteKit (Svelte 5 runes), TypeScript, Vitest, Chart.js (via `ChartCanvas`). Tests under `ui/`; run `cd ui && npx vitest run <path>`. UI composition is verified with `svelte-check` + a production build against a DB copy (the digest page hydration-crashes under `npm run dev` — see [[project_digest_dev_hydration_crash]]).

**Spec:** `docs/superpowers/specs/2026-09-28-digest-2.0-design.md` (§5.2 "The Stats/Graph section"; this phase realizes the "one consolidated Stats section + teaser" IA agreed 2026-10-01).

- **No hardcoded hosts** (the `hostLiterals.test.ts` guard must stay green).

```svelte
<script lang="ts">
	import { flip } from 'svelte/animate';
	import type { Snapshot } from '$lib/server/boarzStats/data.js';

	let { snap }: { snap: Snapshot } = $props();

	const comps = snap.competitors;
	const colors = snap.colors;
	const spansSeasons = new Set(snap.rounds.map((round) => round.seasonNumber)).size > 1;
	const roundLabel = (round: (typeof snap.rounds)[number]) =>
		`${spansSeasons ? `S${round.seasonNumber} · ` : ''}${round.name}`;

	// --- leaderboard race (moved verbatim from LeaderboardTab) -----------------
	const raceRows = $derived(
		comps
			.map((c, i) => ({ c, i, cum: snap.cumulative.map((r) => r[i] ?? 0) }))
			.sort((a, b) => b.cum[b.cum.length - 1] - a.cum[a.cum.length - 1] || a.c.name.localeCompare(b.c.name))
	);
	const R = snap.rounds.length;
	const maxTotal = Math.max(...raceRows.map((r) => r.cum[R - 1]), 1);

	let raceIdx = $state(R - 1);
	let racePlaying = $state(false);
	let fixedRankSlots = $state(false);
	let raceTimer: ReturnType<typeof setInterval> | null = null;

	function raceRank(pi: number, r: number): number {
		const totals = raceRows.map((row) => ({ t: row.cum[r] ?? 0, comp: row.i }));
		const sorted = [...totals].sort((a, b) => b.t - a.t || a.comp - b.comp);
		return sorted.findIndex((o) => o.comp === pi) + 1;
	}
	function stopRace() {
		if (raceTimer) { clearInterval(raceTimer); raceTimer = null; }
		racePlaying = false;
	}
	function toggleRace() {
		if (racePlaying) { stopRace(); return; }
		if (raceIdx >= R - 1) raceIdx = 0;
		racePlaying = true;
		raceTimer = setInterval(() => {
			if (raceIdx >= R - 1) { stopRace(); return; }
			raceIdx += 1;
		}, 1600);
	}
	$effect(() => () => stopRace());

	const raceDisplayRows = $derived(fixedRankSlots
		? [...raceRows].sort((a, b) => (b.cum[raceIdx] ?? 0) - (a.cum[raceIdx] ?? 0) || a.c.name.localeCompare(b.c.name))
		: raceRows);
</script>
```

Create `ui/src/lib/boarzStats/visibleTabs.ts`:

```ts
export interface TabLike {
	id: string;
	seasonOnly?: boolean;
	needsChat?: boolean;
}

export interface VisibleTabOpts {
	hasChat: boolean;
	isAllSeasons: boolean;
	crossSeasonView: 'combined' | 'hide-season';
	omitTabs: readonly string[];
}

/**
 * Which explorer tab ids are visible. Mirrors the StatsExplorer `visibleTabs`
 * rule, plus an explicit `omitTabs` list (used by the digest2 embedding to drop
 * tabs it relocates, e.g. the chat tab). Pure — no DOM, no URL.
 */
export function computeVisibleTabIds(tabs: readonly TabLike[], opts: VisibleTabOpts): string[] {
	const omit = new Set(opts.omitTabs);
	return tabs
		.filter((tab) => (!tab.needsChat || opts.hasChat))
		.filter((tab) => (!tab.seasonOnly || !opts.isAllSeasons || opts.crossSeasonView === 'combined'))
		.filter((tab) => !omit.has(tab.id))
		.map((tab) => tab.id);
}
```
